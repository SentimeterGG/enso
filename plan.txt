func get_absolute_path_from_content_uri(uri_string: String) -> String:
	# 1. Read the bytes from the content URI using your existing method
	var file_bytes: PackedByteArray = read_content_uri(uri_string)
	
	if file_bytes.is_empty():
		push_error("Failed to read data from content URI.")
		return ""
		
	# 2. Define a temporary filename inside Godot's safe user:// directory
	# (You can parse the extension from the URI or default to a generic one)
	var temp_godot_path = "user://temp_shared_file.dat"
	
	# 3. Write the bytes to the temporary file
	var file = FileAccess.open(temp_godot_path, FileAccess.WRITE)
	if not file:
		push_error("Failed to create temporary file at: " + temp_godot_path)
		return ""
		
	file.store_buffer(file_bytes)
	file.close() # Always close the file to flush it to disk
	
	# 4. Convert the internal user:// path into a real OS absolute path
	# On Android, this usually looks like: /data/user/0/your.package.name/files/temp_shared_file.dat
	var absolute_os_path = ProjectSettings.globalize_path(temp_godot_path)
	
	return absolute_os_path
